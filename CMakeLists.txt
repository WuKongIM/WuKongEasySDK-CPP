cmake_minimum_required(VERSION 3.20)
project(WuKongEasySDK VERSION 0.1.0 LANGUAGES CXX)
if(CMAKE_SOURCE_DIR STREQUAL PROJECT_SOURCE_DIR)
    set(WUKONG_TOP_LEVEL ON)
else()
    set(WUKONG_TOP_LEVEL OFF)
endif()
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
option(WUKONG_BUILD_TESTS "Build the SDK tests" ${WUKONG_TOP_LEVEL})
option(WUKONG_BUILD_EXAMPLES "Build the console example" ${WUKONG_TOP_LEVEL})
option(WUKONG_FETCH_JSON "Fetch pinned nlohmann/json when no installed package is available" ON)
option(WUKONG_SANITIZE "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)

find_package(Boost 1.74 REQUIRED CONFIG)
find_package(OpenSSL 1.1.1 REQUIRED)
find_package(Threads REQUIRED)
find_package(nlohmann_json 3.11 QUIET CONFIG)
if(NOT nlohmann_json_FOUND AND WUKONG_FETCH_JSON)
    include(FetchContent)
    set(JSON_Install ON CACHE BOOL "Install JSON dependency")
    FetchContent_Declare(nlohmann_json
        GIT_REPOSITORY https://github.com/nlohmann/json.git
        GIT_TAG 9cca280a4d0ccf0c08f47a99aa71d1b0e52f8d03)
    FetchContent_MakeAvailable(nlohmann_json)
elseif(NOT nlohmann_json_FOUND)
    message(FATAL_ERROR "Install nlohmann_json >= 3.11 or enable WUKONG_FETCH_JSON")
endif()

add_library(wukong_easy_sdk STATIC src/wkim.cpp src/protocol.cpp src/transport.cpp)
add_library(WuKongEasySDK::WuKongEasySDK ALIAS wukong_easy_sdk)
set_target_properties(wukong_easy_sdk PROPERTIES EXPORT_NAME WuKongEasySDK POSITION_INDEPENDENT_CODE ON)
target_compile_features(wukong_easy_sdk PUBLIC cxx_std_17)
target_include_directories(wukong_easy_sdk PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_link_libraries(wukong_easy_sdk PUBLIC nlohmann_json::nlohmann_json
    PRIVATE Boost::headers OpenSSL::SSL OpenSSL::Crypto Threads::Threads)
if(WIN32)
    target_link_libraries(wukong_easy_sdk PRIVATE ws2_32 mswsock)
    target_compile_definitions(wukong_easy_sdk PRIVATE _WIN32_WINNT=0x0601)
endif()
if(MSVC)
    target_compile_options(wukong_easy_sdk PRIVATE /W4)
    target_compile_options(wukong_easy_sdk PUBLIC /utf-8)
else()
    target_compile_options(wukong_easy_sdk PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(WUKONG_SANITIZE AND NOT MSVC)
    target_compile_options(wukong_easy_sdk PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(wukong_easy_sdk PUBLIC -fsanitize=address,undefined)
endif()

install(TARGETS wukong_easy_sdk EXPORT WuKongEasySDKTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT WuKongEasySDKTargets NAMESPACE WuKongEasySDK:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/WuKongEasySDK)
configure_package_config_file(cmake/WuKongEasySDKConfig.cmake.in
    ${CMAKE_CURRENT_BINARY_DIR}/WuKongEasySDKConfig.cmake
    INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/WuKongEasySDK)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/WuKongEasySDKConfigVersion.cmake
    VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/WuKongEasySDKConfig.cmake
    ${CMAKE_CURRENT_BINARY_DIR}/WuKongEasySDKConfigVersion.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/WuKongEasySDK)
if(WUKONG_BUILD_EXAMPLES)
    add_executable(wukong_chat examples/chat.cpp)
    target_link_libraries(wukong_chat PRIVATE WuKongEasySDK::WuKongEasySDK)
endif()
if(WUKONG_BUILD_TESTS)
    enable_testing()
    add_subdirectory(tests)
endif()
