cmake_minimum_required(VERSION 3.20)
project(WuKongPrebuiltConsumer LANGUAGES CXX)
find_package(WuKongEasySDK 0.1 CONFIG REQUIRED)
add_executable(wukong_example main.cpp)
target_link_libraries(wukong_example PRIVATE WuKongEasySDK::WuKongEasySDK)
add_executable(wukong_chat chat.cpp)
target_link_libraries(wukong_chat PRIVATE WuKongEasySDK::WuKongEasySDK)
enable_testing()
add_test(NAME installed_sdk_lifecycle COMMAND wukong_example)
set_tests_properties(installed_sdk_lifecycle PROPERTIES TIMEOUT 10)
